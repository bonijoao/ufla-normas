% Override hyperref's \bibcite to store plain text in b@labels.
% Reason: \@input{aux} inside \document runs BEFORE \AtBeginDocument hooks, so
% we must redefine \bibcite here in the preamble (not via \AtBeginDocument).
% Without this, hyperref's \bibcite stores \hyper@@link in b@key++YEAR, which
% causes abntex2cite's {\yearstyle\@citeyear\@} to fail with
% "Use of \@ doesn't match its definition" in the third pdflatex pass.
\makeatletter
\def\bibcite#1#2{\@newl@bel{b}{#1}{#2}}
\makeatother

% Patch \@lbibitem so abntex2cite's \abntnextkey is set even when hyperref
% overwrites \@lbibitem (hyperref is loaded after abntex2cite, so its
% definition wins; we add \gdef\abntnextkey{#2} to restore the tracking).
\makeatletter
\let\abnt@orig@lbibitem\@lbibitem
\def\@lbibitem[#1]#2{\gdef\abntnextkey{#2}\abnt@orig@lbibitem[{#1}]{#2}}
\makeatother
